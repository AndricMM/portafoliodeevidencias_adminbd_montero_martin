Table cliente {
  id_cliente int [primary key, increment]
  dni_clientes varchar (50)
  nombre_completo varchar
  telefono_principal varchar
  correo varchar
}

Table mascota {
  id_mascota integer [primary key, increment]
  nombre varchar [not null]
  especie varchar
  raza varchar
  fecha_nacimiento date
  id_cliente integer
}

Table veterinario {
  id_veterinario integer [primary key]
  id_cliente int [not null]
  dni varchar (20) [not null, unique]
  nombre varhcar (100) [not null]
  especialidad varchar (100) [not null]
  telefonico varchar (20) [not null]
}

Table cita {
  id_cita integer [primary key]
  fecha datetime [not null]
  diagnostico varchar (50) [not null]
  costo decimal (10,2) [not null]

  id_mascota integer
  id_veterinario integer
}

Table medicamento {
  id_medicamento integer [primary key]
  nombre_medicamento varchar
  fabricante varchar
  precio decimal
}

Table prescripcion_medica {
  id_prescripcion integer [primary key]
  cantidad varchar
  indicaciones text
  id_medicamento integer
  id_cita integer
}
