// Centro de Rescate de Fauna Silvestre "Selva Viva"

Table usuarios {
  id_usuario int [pk, increment]
  nombre varchar(100) [not null]
  correo varchar(100) [not null, unique]
}

Table especies {
  id_especie int [pk, increment]
  nombre_comun varchar(100) [not null]
  nombre_cientifico varchar(120) [not null, unique]
  bioma varchar(50) [not null]
  habito varchar(20) [not null]
}

Table areas {
  id_area int [pk, increment]
  nombre varchar(100) [not null, unique]
  bioma varchar(50) [not null]
  capacidad_maxima int [not null]
  ocupacion_actual int [not null]
}

Table animales {
  id_animal int [pk, increment]
  id_especie int [not null]
  id_area int
  nombre_identificador varchar(50) [not null, unique]
  fecha_ingreso datetime [not null]
  estado_salud varchar(20) [not null]
  estatus varchar(20) [not null]
}

Table diagnosticos {
  id_diagnostico int [pk, increment]
  id_animal int [not null]
  id_veterinario int [not null]
  fecha datetime [not null]
  descripcion varchar(250) [not null]
  tratamiento varchar(250)
  estado_salud varchar(20) [not null]
}

Table movimientos {
  id_movimiento int [pk, increment]
  id_animal int [not null]
  id_usuario_autoriza int [not null]
  tipo varchar(30) [not null]
  origen varchar(150) [not null]
  destino varchar(150) [not null]
  fecha datetime [not null]
  motivo varchar(250) [not null]
}

Ref animales_especie: animales.id_especie > especies.id_especie
Ref animales_area: animales.id_area > areas.id_area
Ref diagnosticos_animal: diagnosticos.id_animal > animales.id_animal
Ref diagnosticos_vet: diagnosticos.id_veterinario > usuarios.id_usuario
Ref movimientos_animal: movimientos.id_animal > animales.id_animal
Ref movimientos_usuario: movimientos.id_usuario_autoriza > usuarios.id_usuario
