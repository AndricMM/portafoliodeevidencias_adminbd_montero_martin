Table users {
  id_user integer [primary key]
  nombre_user varchar
  gmail_user varchar
}

Table libros {
  id_libro integer [primary key]
  titulo_libro varchar
  autor_libro varchar
}

Table prestamos {
  id_prestamos integer [primary key]
  fecha_prestamo datetime
  user_id integer
  libro_id integer
}
